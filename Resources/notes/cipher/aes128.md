# AES-128 (ECB, tự cài đặt)

> Mã khối 128-bit, khóa 128-bit, **10 vòng**. Nhận diện chủ yếu qua **S-box 256 byte** (`63 7C 77 7B F2 6B 6F C5 ...`). Trong CTF có hai dạng: **tự cài đặt** (trang này) và **gọi thư viện** (OpenSSL/BCrypt/CryptoAPI, xem cuối).

## 1. Dấu hiệu nhận biết

- **S-box**: 256 byte bắt đầu `63 7C 77 7B F2 6B 6F C5 30 01 67 2B FE D7 AB 76`. Tìm bằng Search → Bytes (`63 7C 77 7B F2 6B`) hoặc FindCrypt/capa/DIE. **Inverse S-box** (giải mã) bắt đầu `52 09 6A D5 30 36 A5 38`.
- **Hằng `0x1B`** trong phép nhân `xtime`: `(x << 1) ^ ((x >> 7) * 0x1B)`; IDA hiện `(2 * x) ^ (x >> 7) & 0x1B` hoặc `((char)x >> 7) & 0x1B ^ (2 * x)`.
- **Mở rộng khóa**: mảng khóa vòng **176 byte** (`44` từ, dòng `v2 != 44`), điều kiện `i % 16 == 0` / `(4 * i) & 0xF` dùng RotWord+SubWord+**Rcon** (`01 02 04 08 10 20 40 80 1B 36`; Rcon sinh bằng `xtime`).
- **Cấu trúc vòng**: 10 vòng (`v5 == 10`), vòng cuối **bỏ MixColumns**. Trạng thái 16 byte (`__m128i`/mảng 16).
- **ShiftRows** hiện thành chỉ số `((col + row) & 3) * 4 + row`; **MixColumns** là các biểu thức XOR của `xtime(a_i) ^ a_j ...`.
- Giải mã: **Inverse S-box**, `InvMixColumns` nhân với **9, 11, 13, 14** (các hằng `9 11 13 14` hoặc vòng `v19 = 14; v19 >>= 1`).
- Biến thể bảng T (tối ưu): 4 bảng `Te0..Te3` mỗi bảng 1 KB, hằng đầu `0xC66363A5`, `0xF87C7C84`; giải mã `Td0..` đầu `0x51F4A750`.
- Phần cứng: lệnh `aesenc`, `aesenclast`, `aesdec`, `aeskeygenassist` (IDA: `_mm_aesenc_si128`...).

## 2. Thuật toán (mã giả)

```
KeyExpansion(key[16]) -> rk[176]:
    rk[0..15] = key;  rc = 1
    for i = 16; i < 176; i += 4:
        t = rk[i-4 .. i-1]
        nếu i % 16 == 0:  t = SubWord(RotWord(t)) ^ (rc,0,0,0);  rc = xtime(rc)
        rk[i..i+3] = rk[i-16..i-13] ^ t

Encrypt(block):
    s = block ^ rk[0..15]
    for r = 1..10:
        s = SubBytes(s);  s = ShiftRows(s)
        nếu r < 10: s = MixColumns(s)
        s ^= rk[16r .. 16r+15]
Decrypt:   ngược lại: InvShiftRows, InvSubBytes, AddRoundKey, InvMixColumns (trừ vòng đầu tiên của giải mã)
```

## 3. Mã hóa / giải mã

ECB: mỗi khối 16 byte độc lập; **CBC/CTR/GCM** thêm IV/nonce. Quy trình khi gặp trong rev:

1. Xác định chế độ (có `iv`? có XOR với khối trước? tăng bộ đếm?).
2. Lấy **key** (16 byte; có thể chỉ 16 ký tự ASCII) và **IV** từ binary.
3. Giải bằng `Crypto.Cipher.AES` hoặc hàm Python thuần bên dưới.

## 4. Mã nguồn gốc (C)

```c
void aes128_key_expand(const uint8_t key[16], uint8_t rk[176])
{
    uint8_t rc = 1;
    memcpy(rk, key, 16);
    for (int i = 16; i < 176; i += 4) {
        uint8_t t[4];
        memcpy(t, rk + i - 4, 4);
        if (i % 16 == 0) {
            uint8_t u = t[0];
            t[0] = AES_SBOX[t[1]] ^ rc;
            t[1] = AES_SBOX[t[2]];
            t[2] = AES_SBOX[t[3]];
            t[3] = AES_SBOX[u];
            rc = xt(rc);
        }
        for (int j = 0; j < 4; j++) rk[i + j] = rk[i + j - 16] ^ t[j];
    }
}

void aes128_encrypt_block(const uint8_t rk[176], const uint8_t in[16], uint8_t out[16])
{
    uint8_t s[16], t[16];
    for (int i = 0; i < 16; i++) s[i] = in[i] ^ rk[i];
    for (int r = 1; r <= 10; r++) {
        for (int c = 0; c < 4; c++)
            for (int rr = 0; rr < 4; rr++)
                t[c * 4 + rr] = AES_SBOX[s[((c + rr) & 3) * 4 + rr]];
        if (r < 10) {
            for (int c = 0; c < 4; c++) {
                uint8_t a0 = t[c * 4], a1 = t[c * 4 + 1], a2 = t[c * 4 + 2], a3 = t[c * 4 + 3];
                s[c * 4 + 0] = (uint8_t)(xt(a0) ^ (xt(a1) ^ a1) ^ a2 ^ a3);
                s[c * 4 + 1] = (uint8_t)(a0 ^ xt(a1) ^ (xt(a2) ^ a2) ^ a3);
                s[c * 4 + 2] = (uint8_t)(a0 ^ a1 ^ xt(a2) ^ (xt(a3) ^ a3));
                s[c * 4 + 3] = (uint8_t)((xt(a0) ^ a0) ^ a1 ^ a2 ^ xt(a3));
            }
        } else {
            memcpy(s, t, 16);
        }
        for (int i = 0; i < 16; i++) s[i] ^= rk[r * 16 + i];
    }
    memcpy(out, s, 16);
}

void aes128_decrypt_block(const uint8_t rk[176], const uint8_t in[16], uint8_t out[16])
{
    uint8_t s[16], t[16];
    for (int i = 0; i < 16; i++) s[i] = in[i] ^ rk[160 + i];
    for (int r = 9; r >= 0; r--) {
        for (int c = 0; c < 4; c++)
            for (int rr = 0; rr < 4; rr++)
                t[c * 4 + rr] = AES_INV_SBOX[s[((c - rr + 4) & 3) * 4 + rr]];
        for (int i = 0; i < 16; i++) t[i] ^= rk[r * 16 + i];
        if (r > 0) {
            for (int c = 0; c < 4; c++) {
                uint8_t a0 = t[c * 4], a1 = t[c * 4 + 1], a2 = t[c * 4 + 2], a3 = t[c * 4 + 3];
                s[c * 4 + 0] = (uint8_t)(gm(a0, 14) ^ gm(a1, 11) ^ gm(a2, 13) ^ gm(a3, 9));
                s[c * 4 + 1] = (uint8_t)(gm(a0, 9) ^ gm(a1, 14) ^ gm(a2, 11) ^ gm(a3, 13));
                s[c * 4 + 2] = (uint8_t)(gm(a0, 13) ^ gm(a1, 9) ^ gm(a2, 14) ^ gm(a3, 11));
                s[c * 4 + 3] = (uint8_t)(gm(a0, 11) ^ gm(a1, 13) ^ gm(a2, 9) ^ gm(a3, 14));
            }
        } else {
            memcpy(s, t, 16);
        }
    }
    memcpy(out, s, 16);
}
```

```python
def rol8(x, k): return ((x << k) | (x >> (8 - k))) & 0xFF


def _aes_tables():
    sbox = [0] * 256
    p = q = 1
    while True:
        p = (p ^ ((p << 1) & 0xFF) ^ (0x1B if p & 0x80 else 0)) & 0xFF
        q ^= (q << 1) & 0xFF
        q ^= (q << 2) & 0xFF
        q ^= (q << 4) & 0xFF
        if q & 0x80:
            q ^= 0x09
        q &= 0xFF
        x = q ^ rol8(q, 1) ^ rol8(q, 2) ^ rol8(q, 3) ^ rol8(q, 4)
        sbox[p] = (x ^ 0x63) & 0xFF
        if p == 1:
            break
    sbox[0] = 0x63
    inv = [0] * 256
    for i, v in enumerate(sbox):
        inv[v] = i
    return sbox, inv


AES_SBOX, AES_INV_SBOX = _aes_tables()
def _xt(x): return ((x << 1) ^ (0x1B if x & 0x80 else 0)) & 0xFF


def _gm(a, b):
    p = 0
    while b:
        if b & 1:
            p ^= a
        a = _xt(a)
        b >>= 1
    return p


def aes128_expand_key(key):
    rk = bytearray(key)
    rc = 1
    for i in range(16, 176, 4):
        t = list(rk[i - 4:i])
        if i % 16 == 0:
            t = [AES_SBOX[t[1]] ^ rc, AES_SBOX[t[2]], AES_SBOX[t[3]], AES_SBOX[t[0]]]
            rc = _xt(rc)
        rk += bytes(rk[i - 16 + j] ^ t[j] for j in range(4))
    return bytes(rk)


def aes128_decrypt_block(rk, blk):
    s = [blk[i] ^ rk[160 + i] for i in range(16)]
    for r in range(9, -1, -1):
        t = [0] * 16
        for c in range(4):                       # InvShiftRows + InvSubBytes
            for rr in range(4):
                t[c * 4 + rr] = AES_INV_SBOX[s[((c - rr) & 3) * 4 + rr]]
        t = [t[i] ^ rk[r * 16 + i] for i in range(16)]   # AddRoundKey
        if r > 0:                                # InvMixColumns
            s = [0] * 16
            for c in range(4):
                a0, a1, a2, a3 = t[c * 4:c * 4 + 4]
                s[c * 4 + 0] = _gm(a0, 14) ^ _gm(a1, 11) ^ _gm(a2, 13) ^ _gm(a3, 9)
                s[c * 4 + 1] = _gm(a0, 9) ^ _gm(a1, 14) ^ _gm(a2, 11) ^ _gm(a3, 13)
                s[c * 4 + 2] = _gm(a0, 13) ^ _gm(a1, 9) ^ _gm(a2, 14) ^ _gm(a3, 11)
                s[c * 4 + 3] = _gm(a0, 11) ^ _gm(a1, 13) ^ _gm(a2, 9) ^ _gm(a3, 14)
        else:
            s = t
    return bytes(s)


def aes128_ecb_decrypt(key, data):
    rk = aes128_expand_key(key)
    return b"".join(aes128_decrypt_block(rk, data[i:i + 16]) for i in range(0, len(data), 16))
```

## 7. Biến thể & bẫy

- **Gọi thư viện**: `AES_set_encrypt_key`/`AES_encrypt`, `EVP_aes_128_cbc`, `EVP_DecryptInit_ex` (OpenSSL); `BCryptGenerateSymmetricKey`/`BCryptDecrypt` (CNG); `CryptImportKey`/`CryptDecrypt` (CryptoAPI). Khi đó không có S-box trong binary: lấy key/IV/ciphertext từ tham số rồi giải bằng Python.
- **S-box sửa đổi** (đề đổi vài byte hoặc xáo bảng): dùng bảng *đúng như trong binary* khi giải (viết lại hàm với bảng đó).
- **Key schedule bị sửa** hoặc **số vòng khác 10**: đọc pseudo-code.
- AES-192 (12 vòng, khóa 24 byte) và AES-256 (14 vòng, 32 byte): chỉ khác số vòng và mở rộng khóa.
- Padding PKCS#7 ở khối cuối: `unpad` sau khi giải.
